option(AB2_ENABLE_IPO "Enable checked release interprocedural optimization (LTO)" ON)
option(AB2_REQUIRE_IPO "Fail configuration if requested LTO is unavailable" OFF)
if(AB2_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT AB2_IPO_SUPPORTED OUTPUT AB2_IPO_ERROR LANGUAGES CXX)
  if(AB2_IPO_SUPPORTED)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON)
  else()
    if(AB2_REQUIRE_IPO)
      message(FATAL_ERROR "LTO unavailable. Use an LTO-capable compiler (e.g. LLVM-MinGW on Windows).\n${AB2_IPO_ERROR}")
    endif()
    message(WARNING "LTO unavailable: ${AB2_IPO_ERROR}")
  endif()
elseif(AB2_REQUIRE_IPO)
  message(FATAL_ERROR "AB2_REQUIRE_IPO requires AB2_ENABLE_IPO=ON")
endif()

# Match the optimized Make build without changing floating-point semantics or
# requiring the build machine's CPU instruction set on distribution machines.

elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options($<$<CONFIG:Release,RelWithDebInfo>:-O3>)
endif()

      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:${AB2_SDL_TARGET}> $<TARGET_FILE_DIR:afterburner2>)
  endif()

    get_filename_component(AB2_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(runtime libc++.dll libunwind.dll libwinpthread-1.dll)
      if(EXISTS "${AB2_COMPILER_BIN}/${runtime}")
        add_custom_command(TARGET afterburner2 POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E copy_if_different "${AB2_COMPILER_BIN}/${runtime}" $<TARGET_FILE_DIR:afterburner2>)
      endif()
    endforeach()
  endif()
